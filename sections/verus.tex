\begin{frame}[fragile]{Implementing max in Rust}
\begin{lstlisting}
fn max(x: i64, y: i64) -> i64
{
    if x>y {x} else {y}
}
\end{lstlisting}
\end{frame}
\begin{frame}[fragile]{Decorating our implementation with Verus}
\begin{lstlisting}
fn max(x: i64, y: i64) -> (m: i64)
    requires
    ensures
        m>=x,
        m>=y,
        (m==x) || (m==y)
{
    if x>y {x} else {y}
}
\end{lstlisting}
\begin{itemize}
\item \texttt{requires}: what callers must establish.
\item \texttt{ensures}: what callers may rely on afterward.
\end{itemize}
\source{https://verus-lang.github.io/verus/guide/requires_ensures.html}{Verus Guide: contracts}
\note{The complete file is examples/max.rs. There are no preconditions on the two i64 inputs. The executable function only compares and selects, so there is no arithmetic-overflow obligation for its body. The three postconditions characterize the maximum.}
\end{frame}

\begin{frame}{What must be established?}
\begin{enumerate}
\item \textbf{Upper bound:} the result $m$ satisfies $m\ge x$ and $m\ge y$.
\item \textbf{Attainment:} $m=x$ or $m=y$; the result is one of the inputs.
\item \textbf{Both branches:} if $x>y$, return $x$; otherwise $y\ge x$, so return $y$.
\end{enumerate}
\begin{block}{A mutation the contract rejects}
Swap the returned values: \texttt{if x > y \{ y \} else \{ x \}}.\par
For $x=5$, $y=3$, the result $3$ fails $m\ge x$.
\end{block}
\note{No addition occurs in this implementation. The proof is a case split in a total order. At equality either input is a valid maximum, so changing > to >= is not a bug; swapping the returned branches is. The attainment clause matters: an arbitrary upper bound need not be the maximum.}
\end{frame}

\begin{frame}{Using a verified function is applying a lemma}
Let $M(a,b)$ denote the verified function. Its contract gives
\[
 M(a,b)\ge a,\qquad M(a,b)\ge b,\qquad M(a,b)\in\{a,b\}.
\]
To find the maximum of three inputs, compute $t=M(a,b)$, then $m=M(t,c)$.
\begin{itemize}
\item $m\ge t\ge a,b$ and $m\ge c$: $m$ is an upper bound.
\item $m\in\{t,c\}$ and $t\in\{a,b\}$: $m\in\{a,b,c\}$.
\end{itemize}
\begin{block}{Modular proof}
The two contracts establish $m=\max\{a,b,c\}$.\par
We need not inspect either function call's implementation.
\end{block}
\note{All inputs are representable machine integers; comparison and selection stay in the same type. This is ordinary lemma application followed by transitivity. Repeating this argument suggests the maximum-over-a-slice algorithm on the next slides.}
\end{frame}


\begin{frame}{A loop invariant is an induction hypothesis}
For a nonempty slice $a=(a_0,\ldots,a_{n-1})$, after processing $i$ elements:
\[
 I(i,b)\equiv 1\le i\le n\ \land\
 (\forall k<i,\ a_k\le b)\ \land\
 (\exists j<i,\ a_j=b).
\]
Here $j,k$ range over nonnegative integers: $b$ is the maximum of the prefix.
\begin{itemize}
\item \textbf{Base case:} $i=1$, $b=a_0$.
\item \textbf{Inductive step:} replace $b$ by $\max(b,a_i)$, then increment $i$.
\item \textbf{At exit:} $i=n$, so the prefix is the whole slice.
\end{itemize}
\begin{block}{Termination uses well-foundedness}
$V(i)=n-i\in\mathbb{N}$ decreases strictly on every iteration.
\end{block}
\note{The universal clause is an upper bound; the existential clause gives a witness attaining it. Together they specify the maximum rather than just an upper bound. The index bounds justify reading $a_i$ while the guard holds. Initialization reads $a_0$, which is why the input must be nonempty.}
\end{frame}

\begin{frame}{What is being trusted?}
\begin{itemize}
\item The specification captures the intended theorem.
\item The translation from code to logical obligations is sound.
\item The solver's result and any assumed lemmas are correct.
\item Compilation and execution preserve the verified meaning.
\end{itemize}
\begin{block}{Distinguish proof search from proof checking}
Here we rely on the Verus/Z3 verification pipeline.\par
We are not presenting an independently checked proof certificate.
\end{block}
\note{This is a natural question for an audience familiar with proof assistants. Explain the trust boundary of this workflow without implying that SMT solvers cannot produce proofs or that other proof systems have no trusted components. Our demonstration runs verification; it does not export a certificate to a separate small kernel.}
\end{frame}
